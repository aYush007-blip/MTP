\begin{frame}{Stage 1: Decoupled Acoustic Ingestion \& Adaptive VAD}
  \begin{itemize}
    \item \textbf{Decoupled Ingestion Thread:} Non-blocking PortAudio worker captures 16-bit PCM at 16 kHz without blocking downstream processing.
    \item \textbf{Vectorized Decimation:} High-efficiency 48 kHz stereo to 16 kHz mono conversion in NumPy:
    \[ x_{\text{mono}}[n] = \frac{1}{2}\big(x_{\text{left}}[n] + x_{\text{right}}[n]\big) \cdot \frac{1}{32768.0} \]
    \item \textbf{4th-Order Butterworth High-Pass Filter ($f_c = 65$\,Hz):} Eliminates building HVAC rumble and projector fan hum.
    \item \textbf{Adaptive Dual-Rate Energy VAD:} Continuously tracks noise floor $\theta_{\text{noise}}$ over 30ms windows with leaky integration:
    \[ \theta_{\text{noise}}^{(k+1)} = \alpha \theta_{\text{noise}}^{(k)} + (1 - \alpha) E_k \]
    \item \textbf{500ms Pre-Speech Ring Buffer:} Prevents clipping of initial unvoiced mathematical plosives ($/p/$, $/t/$, $/k/$).
  \end{itemize}
\end{frame}

\begin{frame}{The Mathematical Invariant Preservation Principle}
  \begin{block}{Formal Definition}
    \footnotesize
    Let document $\mathcal{D} = \{\mathcal{L}, \mathcal{M}\}$ with discourse $\mathcal{L}$ and math tokens $\mathcal{M}$ in delimiters $\delta \in \{\$, \$\$, \texttt{\textbackslash begin\{equation\}}\}$. Mapping $\mathcal{F}$ is \textbf{Mathematically Invariant} iff:
    \[ \forall m_j \in \mathcal{M}, \quad \mathcal{F}(m_j) \equiv m_j \quad \text{and} \quad \text{Order}\big(\mathcal{F}(m_a)\big) < \text{Order}\big(\mathcal{F}(m_b)\big) \iff a < b \]
  \end{block}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Failure of Standard MT}
        \footnotesize
        Sub-word tokenizers split LaTeX commands (\texttt{\textbackslash f, rac, 1, s+3}), resulting in hallucinated expressions and broken delimiters.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Two-Tier Masking Solution}
        \footnotesize
        1. Extract equations $\to$ replace with tokens (\texttt{\_\_DISPLAY\_EQ\_K\_\_}).\\
        2. Translate surrounding prose into Devanagari.\\
        3. Re-inject identical equation blocks verbatim.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
